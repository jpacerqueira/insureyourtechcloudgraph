def process_data(data):

    for cloud_provider, services in data.items():
        for service in services:
            # Normalize variables
            region = service["Region"].replace(" ", "_").lower()
            availability_zone = service["Availability Zone"].replace(" ", "_").lower()

            # Convert dates to UTC
            impact_date = datetime.strptime(service["Impact Date"], "%Y-%m-%d %H:%M:%S")
            resolved_date = datetime.strptime(service["Resolved Date"], "%Y-%m-%d %H:%M:%S")

            # Format output
            output.append(
                [
                    cloud_provider,
                    service["Service"],
                    region,
                    availability_zone,
                    impact_date.strftime("%Y-%m-%d %H:%M:%S"),
                    service["Duration"],
                    service["Root Cause"],
                    service["Downtime Period"],
                    resolved_date.strftime("%Y-%m-%d %H:%M:%S")
